\pgfplotstableread[col sep=comma]{data/fig07_histograms.csv}\HistogramData

\pgfplotstablecreatecol[
  create col/assign/.code={%
    \edef\MetricName{\thisrow{metric}}%
    \def\SelectorCode{0}%
    \expandafter\IfStrEq\expandafter{\MetricName}{accel}
      {\def\SelectorCode{1}}{}%
    \expandafter\IfStrEq\expandafter{\MetricName}{max_jerk}
      {\def\SelectorCode{2}}{}%
    \expandafter\IfStrEq\expandafter{\MetricName}{max_ang_vel}
      {\def\SelectorCode{3}}{}%
    \expandafter\IfStrEq\expandafter{\MetricName}{avg_ang_vel}
      {\def\SelectorCode{4}}{}%
    \pgfkeyslet{/pgfplots/table/create col/next content}\SelectorCode
  }
]{metric_code}\HistogramData

\newcount\HistogramBinCount
\newcount\HistogramSampleCount
\newcommand{\DeriveHistogramMeta}[2]{%
  \HistogramBinCount=0
  \HistogramSampleCount=0
  \pgfplotstableforeachcolumnelement{metric_code}\of{\HistogramData}\as\RowMetricCode{%
    \IfStrEq{\RowMetricCode}{#1}{%
      \pgfplotstablegetelem{\pgfplotstablerow}{median}\of{\HistogramData}%
      \expandafter\xdef\csname #2Median\endcsname{\pgfplotsretval}%
      \pgfplotstablegetelem{\pgfplotstablerow}{count}\of{\HistogramData}%
      \pgfmathtruncatemacro{\RowSampleCount}{\pgfplotsretval}%
      \global\advance\HistogramBinCount by 1
      \global\advance\HistogramSampleCount by \RowSampleCount
    }{}%
  }%
  \expandafter\xdef\csname #2BinN\endcsname{\the\HistogramBinCount}%
  \expandafter\xdef\csname #2SampleN\endcsname{\the\HistogramSampleCount}%
}
\DeriveHistogramMeta{1}{Accel}
\DeriveHistogramMeta{2}{Jerk}
\DeriveHistogramMeta{3}{MaxAng}
\DeriveHistogramMeta{4}{AvgAng}
\typeout{FIG07_META accel_bins=\AccelBinN, accel_sum=\AccelSampleN,
  accel_median=\AccelMedian}
\typeout{FIG07_META jerk_bins=\JerkBinN, jerk_sum=\JerkSampleN,
  jerk_median=\JerkMedian}
\typeout{FIG07_META max_ang_bins=\MaxAngBinN, max_ang_sum=\MaxAngSampleN,
  max_ang_median=\MaxAngMedian}
\typeout{FIG07_META avg_ang_bins=\AvgAngBinN, avg_ang_sum=\AvgAngSampleN,
  avg_ang_median=\AvgAngMedian}

\newcommand{\DrawHistogramBins}[3]{%
  \pgfplotstableforeachcolumnelement{metric_code}\of{\HistogramData}\as\RowMetricCode{%
    \IfStrEq{\RowMetricCode}{#1}{%
    \pgfplotstablegetelem{\pgfplotstablerow}{left}\of{\HistogramData}\xdef\BinLeft{\pgfplotsretval}
    \pgfplotstablegetelem{\pgfplotstablerow}{right}\of{\HistogramData}\xdef\BinRight{\pgfplotsretval}
    \pgfplotstablegetelem{\pgfplotstablerow}{count}\of{\HistogramData}\xdef\BinCount{\pgfplotsretval}
    \edef\DrawOneBin{%
      \noexpand\path[draw=#2,fill=#2!#3,line width=0.45pt,
        postaction={pattern={Lines[angle=45,distance=4pt,line width=0.25pt]},
          pattern color=#2!70}]
        (axis cs:\BinLeft,0) rectangle (axis cs:\BinRight,\BinCount);}%
    \DrawOneBin
    }{}%
  }%
}

  }
