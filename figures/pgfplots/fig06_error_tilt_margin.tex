\newcommand{\DeriveECDFMeta}[2]{%
  \pgfplotstablegetrowsof{#2}%
  \expandafter\edef\csname #1N\endcsname{\pgfplotsretval}%
  \pgfmathtruncatemacro{\LastIndex}{\pgfplotsretval-1}%
  \expandafter\edef\csname #1LastIndex\endcsname{\LastIndex}%
  \pgfplotstablegetelem{0}{value}\of{#2}%
  \expandafter\edef\csname #1FirstX\endcsname{\pgfplotsretval}%
  \pgfplotstablegetelem{0}{cdf}\of{#2}%
  \expandafter\edef\csname #1FirstCDF\endcsname{\pgfplotsretval}%
  \pgfplotstablegetelem{\LastIndex}{value}\of{#2}%
  \expandafter\edef\csname #1LastX\endcsname{\pgfplotsretval}%
  \pgfplotstablegetelem{\LastIndex}{cdf}\of{#2}%
  \expandafter\edef\csname #1LastCDF\endcsname{\pgfplotsretval}%
}
\DeriveECDFMeta{Error}{\ErrorECDF}
\DeriveECDFMeta{Tilt}{\TiltECDF}
\DeriveECDFMeta{Margin}{\MarginECDF}
