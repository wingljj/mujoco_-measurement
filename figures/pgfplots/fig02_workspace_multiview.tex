\documentclass[tikz,border=0pt]{standalone}
\usepackage{jmechplots}

% Unicode region names are compared after full expansion.  Non-matching rows
% are discarded as NaN; the source table is never rewritten or pre-filtered.
\newcommand{\WorkspacePlot}[4]{%
  \addplot+[#2,only marks,mark size=2.15pt,
    x filter/.code={%
      \edef\RegionInTable{\thisrow{region_cn}}%
      \edef\RegionRequested{#1}%
      \ifx\RegionInTable\RegionRequested\else
        \def\pgfmathresult{nan}%
      \fi
    }]
    table[x=#3,y=#4,col sep=comma]{data/fig02_workspace.csv};%
}

\pgfplotsset{
  workspace/common/.style={
    jmech/base,
    width=4.63cm,height=4.00cm,
    axis equal image,
    xmin=-1.05,xmax=1.95,
    tick label style={font=\fontsize{8}{9.6}\selectfont},
    label style={font=\fontsize{8}{9.6}\selectfont},
    title style={font=\fontsize{8}{9.6}\selectfont,yshift=-1pt},
    legend image post style={mark size=2.15pt},
    clip marker paths=true
  },
  region standard/.style={color=JBlue,mark=o,
    mark options={draw=JBlue,fill=white,line width=0.85pt}},
  region wide/.style={color=JRed,mark=square,
    mark options={draw=JRed,fill=white,line width=0.85pt}},
  region low/.style={color=JGreen,mark=triangle,
    mark options={draw=JGreen,fill=white,line width=0.85pt}},
  region high/.style={color=JGold,mark=diamond,
    mark options={draw=JGold,fill=white,line width=0.85pt}},
  region far/.style={color=black,mark=pentagon,
    mark options={draw=black,fill=white,line width=0.85pt}}
}
